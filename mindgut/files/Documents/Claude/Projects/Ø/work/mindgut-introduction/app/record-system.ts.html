export function distance(a: Triple, b: Triple) { return Math.hypot(a[0] - b[0], a[1] - b[1], a[2] - b[2]); }
export function normalize(values: Triple): Triple {
  const positive = values.map(v => Math.max(0, Number.isFinite(v) ? v : 0));
  const sum = positive[0] + positive[1] + positive[2];
  return sum > 0 ? positive.map(v => v / sum) as Triple : [...MU];
}
export function scanZ3(text: string): ScanResult {
  const counts: Triple = [0, 0, 0], digits: number[] = [];
  for (const character of text) {
    const code = character.charCodeAt(0);
    const value = code >= 65 && code <= 90 ? code - 64 : code >= 97 && code <= 122 ? code - 96 : code >= 49 && code <= 57 ? code - 48 : 0;
    if (!value) continue;
    const digit = 1 + ((value - 1) % 9); digits.push(digit);
    counts[digit % 3 === 1 ? 0 : digit % 3 === 2 ? 1 : 2]++;
  }
  const n = digits.length;
  const fractions = n ? counts.map(c => c / n) as Triple : null;
  return { n, counts, fractions, delta: fractions ? distance(fractions, MU) : null, digits };
}
function fixture(index: number, bias: PhaseBias): Triple {
  // Fixed integer counts: illustrative sample records, never purported live sessions.
  let state = Math.imul(index + 1, 0x45d9f3b) ^ 0x6d2b79f5;
  const next = () => { state ^= state << 13; state ^= state >>> 17; state ^= state << 5; return (state >>> 0) / 4294967296; };
  const values: Triple = [20 + Math.floor(next() * 34), 23 + Math.floor(next() * 34), 23 + Math.floor(next() * 34)];
  const biased = bias === "147" ? 0 : bias === "258" ? 1 : bias === "369" ? 2 : -1;
  if (biased === 0 || biased === 1 || biased === 2) values[biased] += 30;
  return normalize(values);
}
export function makeRecord(count: number, seedFractions?: Triple, bias: PhaseBias = "balanced"): RecordNode[] {
  const length = Math.round(clamp(count, 0, 10000));
  const nodes: RecordNode[] = [];
  for (let index = 0; index < length; index++) {
    const k = index + 1;
    const measured = index === 0 && !!seedFractions;
    const supplied = measured ? seedFractions! : null;
    const validTriple = supplied && supplied.every(v => Number.isFinite(v) && v >= 0) && Math.abs(supplied.reduce((a, b) => a + b, 0) - 1) < 1e-9;
    const f = measured ? (validTriple ? [...supplied!] as Triple : normalize(seedFractions!)) : fixture(index, bias);
    const delta = distance(f, MU), theta = TAU * k / PITCH, y = k * Y_STEP, r = R0 + RADIAL_GAIN * delta;
    nodes.push({ id: `strand-${k}`, index, k, role: k % 2 ? "carbon" : "silicon", f, fractions: f, delta, clash: 0, theta, y, r, xyz: [r * Math.cos(theta), y, r * Math.sin(theta)], illustrative: !measured, source: measured ? "visitor-derived" : "illustrative" });
  }
  for (let index = 0; index < length; index += 2) {
    if (!nodes[index + 1]) continue;
    const clash = distance(nodes[index].f, nodes[index + 1].f);
    nodes[index].clash = clash; nodes[index + 1].clash = clash;
  }
  return nodes;
}

export type OKLab = [number, number, number];
export function srgbToOKLab(rgb: Triple): OKLab {
  const [r, g, b] = rgb.map(v => v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4);
  const l = Math.cbrt(0.4122214708 * r + 0.5363325363 * g + 0.0514459929 * b);
  const m = Math.cbrt(0.2119034982 * r + 0.6806995451 * g + 0.1073969566 * b);
  const s = Math.cbrt(0.0883024619 * r + 0.2817188376 * g + 0.6299787005 * b);
  return [0.2104542553 * l + 0.7936177850 * m - 0.0040720468 * s, 1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s, 0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s];
}
function linearRGB([L, a, b]: OKLab): Triple {
  const l = (L + 0.3963377774 * a + 0.2158037573 * b) ** 3;
  const m = (L - 0.1055613458 * a - 0.0638541728 * b) ** 3;
  const s = (L - 0.0894841775 * a - 1.2914855480 * b) ** 3;
  return [4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s, -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s, -0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s];
}
export function oklabToRGB(lab: OKLab): Triple {
  let chroma = 1, rgb = linearRGB(lab);
  // Reduce chroma, never L, when the requested colour leaves the sRGB gamut.
  while (rgb.some(v => v < -0.000001 || v > 1.000001) && chroma > 0.01) { chroma *= 0.94; rgb = linearRGB([lab[0], lab[1] * chroma, lab[2] * chroma]); }
  return rgb.map(v => { const c = clamp(v, 0, 1); return c <= 0.0031308 ? c * 12.92 : 1.055 * c ** (1 / 2.4) - 0.055; }) as Triple;
}

export function nodeLightness(node: RecordNode, total: number, breath = 4, decay = 1, recalled = false) {
  const active = Math.min(total, Math.round(clamp(breath, 0, 7)) * 2);
  if (recalled) return 0.90;
  if (node.index >= total - active) return 0.88 + 0.06 * (node.index - (total - active) + 1) / Math.max(1, active);
  const older = (node.index + 1) / Math.max(1, total - active);
  return 0.38 + 0.27 * older ** clamp(decay, 0.2, 3);
}
export function nodeColor(node: RecordNode, lightness = 0.74, lOnly = false): string {
  let a = 0, b = 0;
  if (!lOnly) node.f.forEach((f, i) => { a += f * CHANNELS[i].lab[1]; b += f * CHANNELS[i].lab[2]; });
  const rgb = oklabToRGB([clamp(lightness, 0.15, 0.97), a, b]);
  return `rgb(${rgb.map(v => Math.round(v * 255)).join(",")})`;
}
export function recallAddresses(nodes: RecordNode[], highlight = 1): number[] {
  if (nodes.length < 15) return [];

  const candidates = [highlight, 7, 13, 19, 25, 31, 37].map(k => Math.min(ceiling, Math.max(1, k)));

export function worldPoint(node: RecordNode, gain = 1): Triple {
  const r = R0 + RADIAL_GAIN * clamp(gain, 0.05, 4) * node.delta;
  return [r * Math.cos(node.theta), node.y, r * Math.sin(node.theta)];
}
export function createProjection(nodes: RecordNode[], options: ProjectionOptions = {}) {
  const width = options.width ?? 800, height = options.height ?? 800, tilt = clamp(options.tilt ?? 0, 0, 90) * Math.PI / 180;
  const gain = options.gain ?? 1, lastY = nodes.at(-1)?.y ?? 0;
  const centerY = (lastY + (nodes.length ? Y_STEP : 0)) / 2;
  const maxR = Math.max(1.5, ...nodes.map(n => R0 + RADIAL_GAIN * clamp(gain, 0.05, 4) * n.delta));
  // Fit the entire unchanged object. Long records deliberately retain their tall aspect ratio.
  const halfY = Math.max(lastY / 2, 0.4);
  const extentY = maxR * Math.cos(tilt) + halfY * Math.sin(tilt);
  const scale = Math.min(width * 0.39 / maxR, height * 0.38 / Math.max(1.1, extentY));
  function point([x, y, z]: Triple) { const yy = y - centerY; return { x: width / 2 + x * scale, y: height / 2 - (z * Math.cos(tilt) + yy * Math.sin(tilt)) * scale, depth: yy * Math.cos(tilt) - z * Math.sin(tilt) }; }
  return { width, height, scale, point, centerY, nodes: nodes.map(node => ({ node, ...point(worldPoint(node, gain)), scale })) as ProjectedNode[] };
}
